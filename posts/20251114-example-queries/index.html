Get the height values of a player.

query = 'trevor lawrence height'
results = circuity.process_query(database, query)
merged_results_dictionary = {
                             key: [dict[key] for dict in results]
                             for key in results[0]
}
result_values_matrix = circuity.transpose_matrix(list(merged_results_dictionary.values()))
print(",".join(merged_results_dictionary.keys()))
for result_values_list in result_values_matrix:
    print(",".join(result_values_list))
